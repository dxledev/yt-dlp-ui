cmake_minimum_required(VERSION 3.21)
project(YtDlp VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.7 REQUIRED COMPONENTS Core Gui Widgets Network DBus)
add_library(ytdlp_core
    src/core/storage.cpp src/core/theme.cpp src/core/theme_source.cpp src/core/theme_watcher.cpp
    src/core/download.cpp src/core/cookies.cpp src/core/runner.cpp src/core/queue.cpp
    src/ipc/client.cpp src/ipc/server.cpp)
target_include_directories(ytdlp_core PUBLIC src)
target_link_libraries(ytdlp_core PUBLIC Qt6::Core Qt6::Gui Qt6::Network)
add_library(ytdlp_ui src/ui/modal.cpp src/ui/modal_motion.cpp src/ui/portal_picker.cpp src/ui/job_delegate.cpp src/ui/cookie_settings.cpp src/ui/window.cpp)
target_link_libraries(ytdlp_ui PUBLIC ytdlp_core Qt6::Widgets Qt6::DBus)
add_executable(yt-dlp-ui src/main.cpp src/cli.cpp)
target_link_libraries(yt-dlp-ui PRIVATE ytdlp_ui)
foreach(target ytdlp_core ytdlp_ui yt-dlp-ui)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(core_tests tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE ytdlp_core Qt6::Test)
    add_test(NAME core COMMAND core_tests)
    add_executable(worker_tests tests/worker_tests.cpp)
    target_link_libraries(worker_tests PRIVATE ytdlp_core Qt6::Test)
    target_compile_definitions(worker_tests PRIVATE SOURCE_DIRECTORY="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME worker COMMAND worker_tests)
    add_executable(ui_tests tests/ui_tests.cpp)
    target_link_libraries(ui_tests PRIVATE ytdlp_ui Qt6::Test)
    add_test(NAME ui COMMAND ui_tests)
    set_tests_properties(ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;YT_DLP_UI_REDUCED_MOTION=1")
    add_test(NAME integration COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.sh $<TARGET_FILE:yt-dlp-ui> ${CMAKE_CURRENT_SOURCE_DIR})
    set_tests_properties(integration PROPERTIES SKIP_RETURN_CODE 77)
    add_test(NAME install COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/install.sh $<TARGET_FILE_DIR:yt-dlp-ui> ${CMAKE_CURRENT_SOURCE_DIR}/scripts/install-desktop.sh)
endif()
include(GNUInstallDirs)
install(TARGETS yt-dlp-ui RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES assets/io.github.yt_dlp_ui.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/io.github.yt_dlp_ui.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
